cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(marketlab_publisher LANGUAGES CXX)
endif()
include(../cmake/MarketLab.cmake)
if(NOT TARGET marketlab_client)
  add_subdirectory("${MARKETLAB_SOURCE_DIR}/client" "${CMAKE_BINARY_DIR}/client" EXCLUDE_FROM_ALL)
endif()

add_library(marketlab_publisher STATIC publisher.cpp)
target_link_libraries(marketlab_publisher PUBLIC marketlab_market_data_link)
add_library(marketlab_demo_source STATIC demo_source.cpp)
target_link_libraries(marketlab_demo_source PUBLIC marketlab_market_data_link)
add_library(marketlab_publisher_cli STATIC app.cpp)
target_link_libraries(marketlab_publisher_cli PUBLIC marketlab_publisher marketlab_demo_source marketlab_client)
add_executable(market_broadcast main.cpp)
target_link_libraries(market_broadcast PRIVATE marketlab_publisher_cli)
foreach(target marketlab_publisher marketlab_demo_source marketlab_publisher_cli market_broadcast)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
