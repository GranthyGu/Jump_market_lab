cmake_minimum_required(VERSION 3.20)
project(marketlab_broadcast LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(marketlab_md market_data/protocol.cpp market_data/io.cpp market_data/service.cpp)
target_include_directories(marketlab_md PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(marketlab_md PUBLIC Threads::Threads)
target_compile_options(marketlab_md PRIVATE -Wall -Wextra -Wpedantic)
add_executable(market_broadcast market_data/main.cpp)
target_link_libraries(market_broadcast PRIVATE marketlab_md)
enable_testing()
add_executable(broadcast_tests market_data/tests.cpp)
target_link_libraries(broadcast_tests PRIVATE marketlab_md)
add_test(NAME broadcast_tests COMMAND broadcast_tests)
add_test(NAME broadcast_demo COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/market_data/smoke_test.sh
    $<TARGET_FILE:market_broadcast> ${CMAKE_CURRENT_SOURCE_DIR}/market_data/example.conf)
set_tests_properties(broadcast_tests broadcast_demo PROPERTIES TIMEOUT 30)
